// Rasteriza og-fuente.html a 1200 × 630. Uso: node og-rasterizar.mjs [salida.png]
// Necesita Playwright con Chromium (no es dependencia del proyecto: es la herramienta de Cowork).
import { chromium } from 'playwright';
import { fileURLToPath } from 'node:url';
import path from 'node:path';
const aqui = path.dirname(fileURLToPath(import.meta.url));
const salida = process.argv[2] ?? path.join(aqui, 'og-image.png');
const br = await chromium.launch();
const pg = await br.newPage({ viewport: { width: 1200, height: 630 }, deviceScaleFactor: 1 });
await pg.goto('file://' + path.join(aqui, 'og-fuente.html'));
await pg.evaluate(() => document.fonts.ready);
const fuentes = await pg.evaluate(() => [...document.fonts].map((f) => `${f.family} ${f.status}`));
if (fuentes.some((f) => !f.endsWith('loaded') && !f.endsWith('unloaded'))) throw new Error('fuente con error: ' + fuentes);
await pg.screenshot({ path: salida, clip: { x: 0, y: 0, width: 1200, height: 630 } });
console.log(salida, fuentes.join(' · '));
await br.close();
